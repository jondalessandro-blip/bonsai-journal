const isSavingRef = useRef(false)
const [showUnsaved, setShowUnsaved] = useState(false)
const pendingNavRef = useRef<any>(null)

const blocker = useBlocker(
  ({ currentLocation, nextLocation }) => isDirty &&!isSavingRef.current && currentLocation.pathname!== nextLocation.pathname
)

useEffect(()=>{
  if (blocker.state === 'blocked') {
    pendingNavRef.current = blocker
    if (!showUnsaved &&!isSavingRef.current) {
      setShowUnsaved(true)
    }
  }
}, [blocker])

// Discard - WORKS ALREADY
const handleDiscard = () => {
  setShowUnsaved(false)
  setIsDirty(false) // IMPORTANT: clear dirty BEFORE proceed
  setTimeout(()=> pendingNavRef.current?.proceed(), 50)
}

// Save & Leave - FIX DOUBLE
const handleSaveAndLeave = async () => {
  if (isSavingRef.current) return
  isSavingRef.current = true

  try {
    // 1. Save without navigation
    const newTree = await saveTreeToDB(formData) // call your API/supabase insert directly, NOT handleSave that navigates

    // 2. IMMEDIATELY mark clean so blocker won't re-trigger
    setIsDirty(false)

    setShowUnsaved(false)

    // 3. Cancel blocker, don't proceed - manually navigate
